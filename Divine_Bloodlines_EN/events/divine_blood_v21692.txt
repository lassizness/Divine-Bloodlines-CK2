character_event = {
 id = DBV2.1145 hide_window = yes is_triggered_only = yes
 trigger = {
  is_alive = yes
  NOT = { has_character_flag = db_v21692_canonical_identity_done }
  OR = {
   has_character_flag = zeus_blood_founder
   has_character_flag = odin_blood_founder
  }
 }
 immediate = {
  if = {
   limit = { has_character_flag = zeus_blood_founder }
   set_name = "Zeus"
   culture = greek
   set_graphical_culture = greek
   if = {
    limit = { NOR = { religion = hellenic_pagan religion = hellenic_pagan_reformed } }
    religion = hellenic_pagan
   }
  }
  else_if = {
   limit = { has_character_flag = odin_blood_founder }
   set_name = "Odin"
   culture = norse
   set_graphical_culture = norse
   if = {
    limit = { NOR = { religion = norse_pagan religion = norse_pagan_reformed } }
    religion = norse_pagan
   }
  }
  set_character_flag = db_v21692_canonical_identity_done
 }
}
