# Religion lock. FULL gods and active demigods are permanently bound to one pantheon.
# Faded/latent descendants are deliberately NOT included and may freely change religion.
# If active divine blood reawakens, DBV2.10 calls this event immediately and restores the pantheon faith.
# Reformed faith always replaces its unreformed predecessor once the reformation has happened.
character_event = {
 id = DBV2.70 hide_window = yes is_triggered_only = yes
 trigger = { OR = { db_has_active_divine_blood_trigger = yes db_is_active_demigod_trigger = yes } }
 immediate = {
  # Seeing a reformed faith also records the world-level milestone, including reforms made outside our own mission chain.
  if = { limit = { religion = hellenic_pagan_reformed } set_global_flag = db_hellenic_reformed_announced set_character_flag = db_pref_hellenic_reformed }
  if = { limit = { religion = norse_pagan_reformed } set_global_flag = db_norse_reformed_announced set_character_flag = db_pref_norse_reformed }

  # Characters with only one active pantheon are unambiguous. Independent Olympian
  # blood is Hellenic but is deliberately NOT classified as Zeus blood.
  if = {
   limit = { db_has_active_olympian_blood_trigger = yes NOT = { db_has_active_odin_blood_trigger = yes } NOT = { db_has_active_zeus_blood_trigger = yes } }
   set_character_flag = db_pantheon_hellenic
   clr_character_flag = db_pantheon_norse
  }
  else_if = {
   limit = { db_has_active_zeus_blood_trigger = yes NOT = { db_has_active_odin_blood_trigger = yes } }
   set_character_flag = db_pantheon_hellenic
   clr_character_flag = db_pantheon_norse
  }
  else_if = {
   limit = { db_has_active_odin_blood_trigger = yes NOT = { db_has_active_zeus_blood_trigger = yes } }
   set_character_flag = db_pantheon_norse
   clr_character_flag = db_pantheon_hellenic
  }
  else = {
   # Cross-pantheon blood follows the paternal divine line and then keeps that allegiance permanently.
   if = {
    limit = { trait = db_stormrune_blood }
    set_character_flag = db_pantheon_hellenic
    clr_character_flag = db_pantheon_norse
   }
   else_if = {
    limit = { trait = db_fate_thunder_blood }
    set_character_flag = db_pantheon_norse
    clr_character_flag = db_pantheon_hellenic
   }
   else_if = {
    limit = { NOR = { has_character_flag = db_pantheon_hellenic has_character_flag = db_pantheon_norse } OR = { religion = hellenic_pagan religion = hellenic_pagan_reformed } }
    set_character_flag = db_pantheon_hellenic
   }
   else_if = {
    limit = { NOR = { has_character_flag = db_pantheon_hellenic has_character_flag = db_pantheon_norse } OR = { religion = norse_pagan religion = norse_pagan_reformed } }
    set_character_flag = db_pantheon_norse
   }
   else_if = {
    limit = { NOR = { has_character_flag = db_pantheon_hellenic has_character_flag = db_pantheon_norse } father_even_if_dead = { db_has_active_zeus_blood_trigger = yes NOT = { db_has_active_odin_blood_trigger = yes } } }
    set_character_flag = db_pantheon_hellenic
   }
   else_if = {
    limit = { NOR = { has_character_flag = db_pantheon_hellenic has_character_flag = db_pantheon_norse } father_even_if_dead = { db_has_active_odin_blood_trigger = yes NOT = { db_has_active_zeus_blood_trigger = yes } } }
    set_character_flag = db_pantheon_norse
   }
   else_if = {
    limit = { NOR = { has_character_flag = db_pantheon_hellenic has_character_flag = db_pantheon_norse } has_character_flag = db_odin_line_head }
    set_character_flag = db_pantheon_norse
   }
   else_if = {
    limit = { NOR = { has_character_flag = db_pantheon_hellenic has_character_flag = db_pantheon_norse } }
    set_character_flag = db_pantheon_hellenic
   }
  }

  # Enforce the actual faith. The on-conversion hook makes voluntary/forced conversion bounce back immediately;
  # the monthly pulse is a safety net for scripted conversions from other mods/events.
  if = {
   limit = { has_character_flag = db_pantheon_hellenic }
   clr_character_flag = db_pantheon_norse
   if = {
    limit = { OR = { has_global_flag = db_hellenic_reformed_announced has_character_flag = db_pref_hellenic_reformed religion = hellenic_pagan_reformed } }
    set_character_flag = db_pref_hellenic_reformed
    if = { limit = { NOT = { religion = hellenic_pagan_reformed } } religion = hellenic_pagan_reformed }
   }
   else = {
    if = { limit = { NOT = { religion = hellenic_pagan } } religion = hellenic_pagan }
   }
  }
  else_if = {
   limit = { has_character_flag = db_pantheon_norse }
   clr_character_flag = db_pantheon_hellenic
   if = {
    limit = { OR = { has_global_flag = db_norse_reformed_announced has_character_flag = db_pref_norse_reformed religion = norse_pagan_reformed } }
    set_character_flag = db_pref_norse_reformed
    if = { limit = { NOT = { religion = norse_pagan_reformed } } religion = norse_pagan_reformed }
   }
   else = {
    if = { limit = { NOT = { religion = norse_pagan } } religion = norse_pagan }
   }
  }
 }
}
