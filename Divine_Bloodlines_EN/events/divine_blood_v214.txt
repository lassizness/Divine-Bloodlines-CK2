# ============================================================
# v2.1.6.3 - reliable one-time divine arrivals
# ============================================================
# The old v2.1.6 controller was rooted on the human player and marked an arrival
# "consumed" before a founder had actually been created. If the delayed character
# event was lost, or any later step failed, the watchdog could permanently suppress
# the god. v2.1.6.3 moves scheduling to the holders of the destination counties.
# Those characters exist independently of who the player controls, so the system
# survives succession, switching characters and lost delayed events.

# Fresh-campaign marker used only to distinguish a newly started v2.1.6.3 game
# from a save upgraded from the broken v2.1.6/2.1.6.2 controller.
character_event = {
 id = DBV2.977 hide_window = yes is_triggered_only = yes
 trigger = { ai = no }
 immediate = { set_global_flag = db_v2163_native_campaign }
}

# Startup reconciliation only. Arrival timing itself is NOT owned by the player.
character_event = {
 id = DBV2.978 hide_window = yes is_triggered_only = yes
 trigger = { ai = no }
 immediate = {
  set_global_flag = db_divine_arrival_watch_started
  if = {
   limit = { OR = { has_global_flag = db_zeus_founder_exists any_character = { has_character_flag = zeus_blood_founder } } }
   set_global_flag = db_zeus_founder_exists
   set_global_flag = db_zeus_arrival_consumed
  }
  if = {
   limit = { OR = { has_global_flag = db_odin_founder_exists any_character = { has_character_flag = odin_blood_founder } } }
   set_global_flag = db_odin_founder_exists
   set_global_flag = db_odin_arrival_consumed
  }
 }
}

# Compatibility entry point retained for old saves/events. It only reconciles stale
# consumed flags; the county-holder schedulers below are authoritative.
character_event = {
 id = DBV2.979 hide_window = yes is_triggered_only = yes
 trigger = { ai = no }
 immediate = {
  if = {
   limit = { has_global_flag = db_zeus_arrival_consumed NOT = { has_global_flag = db_zeus_founder_exists } NOT = { any_character = { has_character_flag = zeus_blood_founder } } }
   clr_global_flag = db_zeus_arrival_consumed
  }
  if = {
   limit = { has_global_flag = db_odin_arrival_consumed NOT = { has_global_flag = db_odin_founder_exists } NOT = { any_character = { has_character_flag = odin_blood_founder } } }
   clr_global_flag = db_odin_arrival_consumed
  }
 }
}

# Zeus scheduler. on_yearly_pulse is character-scoped, but only the current holder
# of Arta can pass this trigger. The first two annual checks are randomized;
# the third is forced. Thus Zeus appears during the first few years without relying
# on the player character at all.
character_event = {
 id = DBV2.986 hide_window = yes is_triggered_only = yes
 trigger = {
  has_landed_title = c_arta
  NOT = { has_global_flag = db_zeus_founder_exists }
  NOT = { any_character = { has_character_flag = zeus_blood_founder } }
 }
 immediate = {
  if = { limit = { has_global_flag = db_zeus_arrival_consumed } clr_global_flag = db_zeus_arrival_consumed }
  if = {
   limit = { NOT = { has_global_flag = db_v2163_zeus_year_1 } }
   set_global_flag = db_v2163_zeus_year_1
   random_list = { 35 = { character_event = { id = DBV2.981 } } 65 = { } }
  }
  else_if = {
   limit = { NOT = { has_global_flag = db_v2163_zeus_year_2 } }
   set_global_flag = db_v2163_zeus_year_2
   random_list = { 60 = { character_event = { id = DBV2.981 } } 40 = { } }
  }
  else = { character_event = { id = DBV2.981 } }
 }
}

  has_landed_title = c_arta

  clr_global_flag = db_zeus_arrival_consumed

# Zeus: create a clean character. Crucially, do NOT mark the arrival consumed until
# after the founder initialization has run. If anything before that fails, the next
# county-holder pulse can retry instead of permanently suppressing Zeus.
# Zeus creation stage. v2.1.6.6 keeps the current holder of Arta in Epirus untouched
# while this event is running. Zeus is initialized landless; title transfer happens
# in DBV2.1060 on the following day.
character_event = {
 id = DBV2.981 hide_window = yes is_triggered_only = yes
 trigger = {
  NOT = { has_global_flag = db_zeus_founder_exists }
  NOT = { any_character = { has_character_flag = zeus_blood_founder } }
 }
 immediate = {
  log = "[DB ARRIVAL] ZEUS_CREATE_BEGIN"
  create_character = {
   name = "Zeus"
   dynasty = 990212001
   religion = hellenic_pagan
   culture = greek
   age = 32
   female = no
   random_traits = no
   historical = yes
  }
  new_character = { save_event_target_as = db_auto_zeus }
  event_target:db_auto_zeus = {
   set_character_flag = db_auto_arrived_zeus
   add_trait = zeus_origin_marker
   character_event = { id = DBV2.1 }
   if = {
    limit = { has_character_flag = zeus_blood_founder }
    set_character_flag = db_v216_fresh_arrival
    log = "[DB ARRIVAL] ZEUS_INITIALIZED"
    character_event = { id = DBV2.1060 days = 1 }
   }
  }
 }
}

# ============================================================
# v2.1.6.4 repair: restore the passive divine allure maintenance event
# accidentally dropped from the 2.1.6.3 arrival rewrite.
# This is silent: it only maintains opinion/sex-appeal modifiers.
# ============================================================
character_event = {
 id = DBV2.990 hide_window = yes is_triggered_only = yes
 trigger = {
  OR = {
   db_is_true_god_trigger = yes
   db_is_active_demigod_trigger = yes
   has_character_modifier = db_true_god_allure
   has_character_modifier = db_demigod_allure
  }
 }
 immediate = {
  if = {
   limit = { db_is_true_god_trigger = yes }
   remove_character_modifier = db_demigod_allure
   remove_character_modifier = db_true_god_allure
   add_character_modifier = { name = db_true_god_allure days = 40 hidden = yes }
  }
  else_if = {
   limit = { db_is_active_demigod_trigger = yes }
   remove_character_modifier = db_true_god_allure
   remove_character_modifier = db_demigod_allure
   add_character_modifier = { name = db_demigod_allure days = 40 hidden = yes }
  }
  else = {
   remove_character_modifier = db_true_god_allure
   remove_character_modifier = db_demigod_allure
  }
 }
}
