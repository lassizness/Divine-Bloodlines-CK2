# ============================================================
# v2.1.5 - founder sanitation / old-save repair
# ============================================================
character_event = {
 id = DBV2.1010 hide_window = yes is_triggered_only = yes
 trigger = { is_alive = yes OR = { has_character_flag = zeus_blood_founder has_character_flag = odin_blood_founder } }
 immediate = {
  # Arrival must never make a god sterile/celibate by inheriting mortal state.
  remove_trait = celibate
  remove_trait = eunuch
  remove_trait = monk
  remove_trait = nun
  remove_trait = weak
  remove_trait = frail
  remove_trait = infirm
  remove_trait = incapable
  cure_illness = yes
  if = { limit = { NOT = { trait = db_severed_immortality } age >= 16 } add_trait = immortal diplomatic_immunity = no  }
  if = {
   limit = { has_character_flag = zeus_blood_founder }
   set_character_flag = db_pantheon_hellenic
   # v2.1.6: auto-arrived Zeus is a territorial ruler, not a landless manifestation.
   if = { limit = { has_character_flag = db_auto_arrived_zeus NOT = { has_character_flag = db_v216_arrival_established } } character_event = { id = DBV2.1060 days = 1 } }
  }
  if = {
   limit = { has_character_flag = odin_blood_founder }
   set_character_flag = db_pantheon_norse
   # v2.1.6: auto-arrived Odin gets a real county and is made independent.
   if = { limit = { has_character_flag = db_auto_arrived_odin NOT = { has_character_flag = db_v216_arrival_established } } character_event = { id = DBV2.1060 days = 1 } }
  }
  # DBV2.70 enforces the correct unreformed/reformed pantheon faith without temporary conversion loops.
  character_event = { id = DBV2.70 }
  character_event = { id = DBV2.990 }
 }
}
