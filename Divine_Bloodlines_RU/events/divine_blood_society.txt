# Seed a leader and a small core after the Sickle legend awakens.
character_event = {
    id = DBH.1
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = db_order_sickle_awakened
        has_dlc = "Mystics"
        OR = {
            NOT = { has_global_flag = db_order_sickle_initialized }
            db_order_sickle = { society_has_members = no }
        }
    }
    immediate = {
        if = { limit = { db_order_sickle = { society_has_members = no } } clr_global_flag = db_order_sickle_initialized }
        random_character = {
            limit = {
                ai = yes
                is_adult = yes
                is_alive = yes
                is_in_society = no
                prisoner = no
                is_incapable = no
                NOT = { db_is_true_god_trigger = yes }
                NOT = { db_is_active_demigod_trigger = yes }
                OR = {
                    intrigue >= 12
                    learning >= 12
                    trait = ambitious
                    trait = cynical
                    trait = deceitful
                }
            }
            join_society = db_order_sickle
            society_rank_up = 3
            set_society_grandmaster = yes
            change_society_currency = 1200
            set_character_flag = db_order_first_master
            set_global_flag = db_order_sickle_initialized
        }

        if = {
            limit = { has_global_flag = db_order_sickle_initialized }
            random_character = {
                limit = { ai = yes is_adult = yes is_alive = yes is_in_society = no prisoner = no NOT = { db_is_true_god_trigger = yes } NOT = { db_is_active_demigod_trigger = yes } }
                join_society = db_order_sickle
                society_rank_up = 1
                change_society_currency = 400
            }
            random_character = {
                limit = { ai = yes is_adult = yes is_alive = yes is_in_society = no prisoner = no NOT = { db_is_true_god_trigger = yes } NOT = { db_is_active_demigod_trigger = yes } }
                join_society = db_order_sickle
                society_rank_up = 1
                change_society_currency = 400
            }
            random_character = {
                limit = { ai = yes is_adult = yes is_alive = yes is_in_society = no prisoner = no NOT = { db_is_true_god_trigger = yes } NOT = { db_is_active_demigod_trigger = yes } }
                join_society = db_order_sickle
                change_society_currency = 250
            }
            random_character = {
                limit = { ai = yes is_adult = yes is_alive = yes is_in_society = no prisoner = no NOT = { db_is_true_god_trigger = yes } NOT = { db_is_active_demigod_trigger = yes } }
                join_society = db_order_sickle
                change_society_currency = 250
            }
        }
    }
}

# Only the current grandmaster recruits. This replaces the old per-character yearly
# lottery that could absorb thousands of AI characters into a secret society.
character_event = {
    id = DBH.2
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = db_order_sickle_awakened
        has_dlc = "Mystics"
        society_member_of = db_order_sickle
        is_society_grandmaster = yes
        is_alive = yes
        db_order_sickle = { num_of_society_members < 45 }
    }
    immediate = {
        if = {
            limit = { db_order_sickle = { num_of_society_members < 15 } }
            character_event = { id = DBH.8 }
            character_event = { id = DBH.8 }
            character_event = { id = DBH.8 }
            character_event = { id = DBH.8 }
        }
        else_if = {
            limit = { db_order_sickle = { num_of_society_members < 30 } }
            character_event = { id = DBH.8 }
            character_event = { id = DBH.8 }
        }
        else = {
            character_event = { id = DBH.8 }
        }
    }
}

            is_incapable = no

            NOT = { has_character_modifier = db_order_recruitment_cooldown }
            NOT = { has_character_modifier = db_order_initiation_pending }
            NOT = { db_is_true_god_trigger = yes }
            NOT = { db_is_active_demigod_trigger = yes }
        }
    }
    immediate = {
        save_event_target_as = dbh_recruiter
        random_character = {
            limit = {
                ai = no
                is_alive = yes
                is_adult = yes
                prisoner = no
                is_incapable = no
                can_join_society = db_order_sickle
                NOT = { has_character_modifier = db_order_recruitment_cooldown }
                NOT = { has_character_modifier = db_order_initiation_pending }
                NOT = { db_is_true_god_trigger = yes }
                NOT = { db_is_active_demigod_trigger = yes }
            }
            if = {
                limit = { db_has_godslayer_blood_trigger = yes }
                character_event = { id = DBH.11 }
            }
            else_if = {
                limit = { OR = { any_friend = { db_is_true_god_trigger = yes } any_lover = { db_is_true_god_trigger = yes } } }
                random = { chance = 10 character_event = { id = DBH.11 } }
            }
            else_if = {
                limit = { OR = { intrigue >= 14 martial >= 14 learning >= 14 trait = brave trait = hunter trait = ambitious trait = deceitful } }
                random = { chance = 65 character_event = { id = DBH.11 } }
            }
            else = { random = { chance = 30 character_event = { id = DBH.11 } } }
        }
    }
}
