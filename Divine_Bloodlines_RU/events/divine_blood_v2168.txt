# v2.1.6.8 - Heroic Path is an earned adulthood event, not an automatic age-16 popup.
character_event = {
 id = DBV2.659 hide_window = yes is_triggered_only = yes
 trigger = {
  is_alive = yes age >= 20 db_is_active_demigod_trigger = yes
  NOT = { has_character_flag = db_hero_path_offered }
  NOR = { has_character_modifier = db_hero_warrior has_character_modifier = db_hero_seeker has_character_modifier = db_hero_guardian has_character_modifier = db_hero_sage has_character_modifier = db_hero_trickster }
  OR = { prestige >= 500 martial >= 12 diplomacy >= 12 intrigue >= 12 learning >= 12 }
 }
 immediate = { random = { chance = 25 character_event = { id = DBV2.660 } } }
}

# Legendary Gathering repair: Holy Fury requires two accessible rank-IV heroes.
# If a true god is already a Hero, promote an available lodge veteran before the progress bar fills.
character_event = {
 id = DBV2.1100 hide_window = yes is_triggered_only = yes
 trigger = {
  is_alive = yes db_is_true_god_trigger = yes society_rank == 4 is_member_of_any_warrior_lodge_trigger = yes
  society = { society_has_active_progress = yes }
  NOT = { has_character_flag = call_to_legend_active }
  NOT = { society = { any_society_member = { count >= 2 society_rank == 4 is_inaccessible_trigger = no } } }
 }
 immediate = {
  society = {
   random_society_member = {
    limit = { ai = yes is_alive = yes is_adult = yes prisoner = no is_incapable = no NOT = { character = ROOT } society_rank < 4 }
    if = { limit = { society_rank == 3 } society_rank_up = 1 }
    else_if = { limit = { society_rank == 2 } society_rank_up = 2 }
    else = { society_rank_up = 3 }
    set_character_flag = db_divine_legend_companion
   }
  }
 }
}

# Legendary Journey protection: a true god entering the top bracket cannot be removed by the vanilla 10% "guards imprison the hero" branch, which only applies below progress 10.
character_event = {
 id = DBV2.1101 hide_window = yes is_triggered_only = yes
 trigger = {
  is_alive = yes db_is_death_guarded_true_god_trigger = yes
  has_character_flag = call_to_legend_active
  has_character_flag = top_bracket
  check_variable = { which = legendary_event_progress value < 10 }
 }
 immediate = { set_variable = { which = legendary_event_progress value = 10 } }
}



# If a Legendary Journey participant dies and no active hero remains, release the warrior lodge state immediately.
character_event = {
 id = DBV2.1103 hide_window = yes is_triggered_only = yes
 trigger = { has_character_flag = call_to_legend_active }
 immediate = {
  if = {
   limit = {
    is_in_society = yes
    NOT = { society = { any_society_member = { is_alive = yes is_dying = no has_character_flag = call_to_legend_active } } }
   }
   society = {
    set_society_progress = 0
    clr_flag = block_society_progress
    clr_flag = used_legendary_gathering
    any_society_member = { limit = { OR = { has_character_flag = call_to_legend_active has_character_flag = completed_call_to_legend } } clean_up_after_legendary_showdown_effect = yes }
   }
  }
 }
}
