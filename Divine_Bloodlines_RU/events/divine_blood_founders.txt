   set_character_flag = db_sickle_legend_personally_warned

  }

# v2.1.2 HOTFIX3 - one-time name normalization for existing saves.
character_event = {
 id = DBV2.960 hide_window = yes is_triggered_only = yes
 trigger = {
  OR = { has_character_flag = zeus_blood_founder has_character_flag = odin_blood_founder }
  NOT = { has_character_flag = db_ru_divine_names_migrated }
 }
 immediate = {
  if = {
   limit = { has_character_flag = zeus_blood_founder }
   set_name = "Зевс"
   father_even_if_dead = { set_name = "Кронос" }
   mother_even_if_dead = { set_name = "Рея" }
   any_spouse = {
    limit = { trait = pure_divine_blood_zeus is_female = yes }
    set_name = "Гера"
    set_character_flag = db_hera
   }
   any_sibling_even_if_dead = {
    if = { limit = { has_character_flag = db_poseidon } set_name = "Посейдон" }
    if = { limit = { has_character_flag = db_hades } set_name = "Аид" }
   }
  }
  if = {
   limit = { has_character_flag = odin_blood_founder }
   set_name = "Один"
   any_spouse = {
    limit = { trait = pure_divine_blood_odin is_female = yes }
    set_name = "Фригг"
    set_character_flag = db_frigg
   }
  }
  set_character_flag = db_ru_divine_names_migrated
 }
}
